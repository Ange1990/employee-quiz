rules_version = '2';
service cloud.firestore {
 match /databases/{database}/documents {
  function signedIn() { return request.auth != null; }
  function admin() { return signedIn() && request.auth.token.email in ['nafpliotis@sspc.gr','nafpliotou@sspc.gr','tzanetopoulou@sspc.gr']; }
  // Original collections: unchanged behavior until production migration.
  match /questions/{id} { allow read, write: if signedIn(); }
  match /results/{id} { allow read, write: if signedIn(); }
  match /users/{id} { allow read, write: if signedIn() && request.auth.uid == id; }
  // New V3 collections
  match /questions_v3/{id} { allow read: if signedIn(); allow create, update, delete: if admin(); }
  match /results_v3/{id} {
   allow create: if signedIn() && (admin() || (request.auth.uid == id && request.resource.data.uid == request.auth.uid && request.resource.data.email == request.auth.token.email && request.resource.data.assessmentVersion == 3));
   allow read: if admin() || (signedIn() && request.auth.uid == id);
   allow update: if false;
   allow delete: if admin();
  }
 }
}
